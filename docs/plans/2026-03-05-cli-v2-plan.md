// src/cli/GlobalOptions.ts
import type { Command } from "commander";

interface GlobalOptions {
  readonly json: boolean;
  readonly verbose: boolean;
  readonly dryRun: boolean;
  readonly noColor: boolean;
  readonly concurrency: number | undefined;
}

function applyGlobalOptions(program: Command): Command {
  return program
    .option("--json", "Force JSON output on all commands", false)
    .option("--verbose", "Show detailed logs", false)
    .option("--dry-run", "Show what would happen without making changes", false)
    .option("--no-color", "Disable colored output")
    .option("--concurrency <n>", "Override default concurrency", parseInt);
}

function parseGlobalOptions(cmd: Command): GlobalOptions {
  const opts = cmd.opts();
  return {
    json: opts.json ?? false,
    verbose: opts.verbose ?? false,
    dryRun: opts.dryRun ?? false,
    noColor: opts.color === false,
    concurrency: opts.concurrency,
  };
}

export { applyGlobalOptions, parseGlobalOptions, type GlobalOptions };
```

// src/cli/output/TableRenderer.ts

interface ColumnDef {
  readonly key: string;
  readonly header: string;
  readonly width: number;
}

interface RenderOptions {
  readonly format: "table" | "csv" | "json";
  readonly total?: number;
  readonly offset?: number;
  readonly limit?: number;
}

function renderTable(
  rows: readonly Record<string, unknown>[],
  columns: readonly ColumnDef[],
  options: RenderOptions
): string {
  if (options.format === "json") {
    return JSON.stringify(rows, null, 2);
  }
  if (options.format === "csv") {
    return renderCsv(rows, columns);
  }
  return renderTextTable(rows, columns, options);
}

function renderCsv(rows: readonly Record<string, unknown>[], columns: readonly ColumnDef[]): string {
  const header = columns.map((c) => c.header).join(",");
  const dataLines = rows.map((row) =>
    columns
      .map((col) => {
        const val = String(row[col.key] ?? "");
        return val.includes(",") || val.includes('"') ? `"${val.replace(/"/g, '""')}"` : val;
      })
      .join(",")
  );
  return [header, ...dataLines].join("\n");
}

function truncate(str: string, width: number): string {
  if (str.length <= width) return str;
  return str.slice(0, width - 3) + "...";
}

function renderTextTable(
  rows: readonly Record<string, unknown>[],
  columns: readonly ColumnDef[],
  options: RenderOptions
): string {
  const pad = (str: string, width: number): string => {
    const truncated = truncate(str, width);
    return truncated + " ".repeat(Math.max(0, width - truncated.length));
  };

  const headerLine = columns.map((c) => pad(c.header, c.width)).join(" | ");
  const separator = columns.map((c) => "-".repeat(c.width)).join("-+-");
  const dataLines = rows.map((row) =>
    columns.map((col) => pad(String(row[col.key] ?? ""), col.width)).join(" | ")
  );

  const parts = [headerLine, separator, ...dataLines];

  if (options.total !== undefined) {
    const start = (options.offset ?? 0) + 1;
    const end = (options.offset ?? 0) + rows.length;
    parts.push("");
    parts.push(`Showing ${start}-${end} of ${options.total} results`);
    if (options.limit !== undefined && end < options.total) {
      parts[parts.length - 1] += ` (use --offset ${end} for next page)`;
    }
  }

  return parts.join("\n");
}

export { renderTable, type ColumnDef, type RenderOptions };
```

// src/cli/output/Progress.ts

interface ProgressOptions {
  readonly total: number;
  readonly label: string;
}

interface ProgressTracker {
  readonly total: number;
  current: number;
  increment(n?: number): void;
  render(): string;
}

function createProgress(options: ProgressOptions): ProgressTracker {
  const tracker: ProgressTracker = {
    total: options.total,
    current: 0,
    increment(n = 1) {
      tracker.current = Math.min(tracker.current + n, options.total);
    },
    render() {
      const pct = Math.round((tracker.current / options.total) * 100);
      const barWidth = 20;
      const filled = Math.round((tracker.current / options.total) * barWidth);
      const bar = "\u2588".repeat(filled) + "\u2591".repeat(barWidth - filled);
      return `  ${options.label} ${bar} ${pct}%  ${tracker.current}/${options.total}`;
    },
  };
  return tracker;
}

interface Spinner {
  message: string;
  update(msg: string): void;
  done(msg: string): string;
  fail(msg: string): string;
}

function createSpinner(message: string): Spinner {
  const spinner: Spinner = {
    message,
    update(msg: string) {
      spinner.message = msg;
    },
    done(msg: string) {
      return statusMessage("success", msg);
    },
    fail(msg: string) {
      return statusMessage("error", msg);
    },
  };
  return spinner;
}

function statusMessage(type: "success" | "error" | "info" | "warn", msg: string): string {
  const PREFIX = {
    success: "v",
    error: "x",
    info: "->",
    warn: "!",
  } as const;
  return `${PREFIX[type]} ${msg}`;
}

export { createProgress, createSpinner, statusMessage };
export type { ProgressOptions, ProgressTracker, Spinner };
```

// src/cli/runCommand.ts

interface RunCommandOptions {
  readonly dryRun?: boolean;
  readonly onError?: (error: unknown) => void;
}

async function runCommand(
  action: () => Promise<void>,
  options: RunCommandOptions = {}
): Promise<number> {
  if (options.dryRun) {
    process.exitCode = 0;
    return 0;
  }
  try {
    await action();
    process.exitCode = 0;
    return 0;
  } catch (error) {
    if (options.onError) {
      options.onError(error);
    } else {
      const msg = error instanceof Error ? error.message : String(error);
      console.error(statusMessage("error", msg));
    }
    process.exitCode = 1;
    return 1;
  }
}

import { statusMessage } from "./output/Progress";

export { runCommand, type RunCommandOptions };
```

// src/cli/groups/bootstrap.ts
import type { Command } from "commander";
import { runWorkflow, runTasks } from "@workglow/cli";
import { FetchAllCikNamesTask } from "../../task/ciknames/FetchAllCikNamesTask";
import { StoreCikNamesTask } from "../../task/ciknames/StoreCikNamesTask";
import { BootstrapDownloadTask } from "../../task/bootstrap/BootstrapDownloadTask";
import { BootstrapSubmissionsTask } from "../../task/submissions/BootstrapSubmissionsTask";
import { BootstrapCompanyFactsTask } from "../../task/facts/BootstrapCompanyFactsTask";
import { pipe } from "@workglow/task-graph";
import { runCommand } from "../runCommand";
import { statusMessage } from "../output/Progress";

const BULK_DOWNLOADS = {
  submissions: {
    url: "https://www.sec.gov/Archives/edgar/daily-index/bulkdata/submissions.zip",
    targetFolder: "submissions",
  },
  facts: {
    url: "https://www.sec.gov/Archives/edgar/daily-index/xbrl/companyfacts.zip",
    targetFolder: "companyfacts",
  },
} as const;

function addBootstrapCommands(parent: Command): void {
  const bootstrap = parent
    .command("bootstrap")
    .description("Download, ingest, and process SEC EDGAR data")
    .option("--skip-download", "Skip download phase", false)
    .option("--skip-ingest", "Skip ingest phase", false)
    .option("--skip-forms", "Skip form processing phase", false)
    .action(async (options) => {
      await runCommand(async () => {
        if (!options.skipDownload) {
          console.log(statusMessage("info", "Phase 1: Downloading bulk data..."));
          for (const [, config] of Object.entries(BULK_DOWNLOADS)) {
            await runTasks(new BootstrapDownloadTask(config));
          }
        }
        if (!options.skipIngest) {
          console.log(statusMessage("info", "Phase 2: Ingesting data..."));
          const cikWf = pipe([new FetchAllCikNamesTask(), new StoreCikNamesTask()]);
          await runWorkflow(cikWf);
          await runTasks(new BootstrapSubmissionsTask());
          await runTasks(new BootstrapCompanyFactsTask());
        }
        if (!options.skipForms) {
          console.log(statusMessage("info", "Phase 3: Processing forms..."));
          // Form processing delegated to UpdateAllFormsTask
          const { UpdateAllFormsTask } = await import("../../task/forms/UpdateAllFormsTask");
          await runTasks(new UpdateAllFormsTask({ form: ["D", "C", "1-A", "1-K", "1-Z"] }));
        }
      });
    });

  bootstrap
    .command("download <type>")
    .description("Download bulk SEC data (submissions | facts | ciks | all)")
    .action(async (type: string) => {
      await runCommand(async () => {
        if (type !== "submissions" && type !== "facts" && type !== "ciks" && type !== "all") {
          throw new Error(`Invalid type "${type}". Must be submissions, facts, ciks, or all.`);
        }
        const types: (keyof typeof BULK_DOWNLOADS)[] =
          type === "all" ? ["submissions", "facts"] : type === "ciks" ? [] : [type as keyof typeof BULK_DOWNLOADS];

        if (type === "ciks" || type === "all") {
          // CIK download handled by FetchAllCikNamesTask
          const cikWf = pipe([new FetchAllCikNamesTask(), new StoreCikNamesTask()]);
          await runWorkflow(cikWf);
        }

        for (const t of types) {
          await runTasks(new BootstrapDownloadTask(BULK_DOWNLOADS[t]));
        }
      });
    });

  bootstrap
    .command("ingest [domain]")
    .description("Ingest downloaded data (submissions | facts | cik-names | all)")
    .action(async (domain: string = "all") => {
      await runCommand(async () => {
        if (domain === "all" || domain === "cik-names") {
          const cikWf = pipe([new FetchAllCikNamesTask(), new StoreCikNamesTask()]);
          await runWorkflow(cikWf);
        }
        if (domain === "all" || domain === "submissions") {
          await runTasks(new BootstrapSubmissionsTask());
        }
        if (domain === "all" || domain === "facts") {
          await runTasks(new BootstrapCompanyFactsTask());
        }
      });
    });
}

export { addBootstrapCommands };

import type { Command } from "commander";

interface InitConfig {
  readonly dbType: "sqlite" | "postgres";
  readonly dbFolder: string;
  readonly dbName: string;
  readonly rawDataFolder: string;
  readonly pgUrl?: string;
  readonly pgHost?: string;
  readonly pgPort?: string;
  readonly pgUser?: string;
  readonly pgPassword?: string;
  readonly pgDatabase?: string;
}

function buildEnvConfig(config: InitConfig): string {
  const lines: string[] = [
    `SEC_DB_TYPE="${config.dbType}"`,
    `SEC_DB_FOLDER="${config.dbFolder}"`,
    `SEC_DB_NAME="${config.dbName}"`,
    `SEC_RAW_DATA_FOLDER="${config.rawDataFolder}"`,
  ];

  if (config.dbType === "postgres") {
    if (config.pgUrl) {
      lines.push(`SEC_PG_URL="${config.pgUrl}"`);
    } else {
      if (config.pgHost) lines.push(`SEC_PG_HOST="${config.pgHost}"`);
      if (config.pgPort) lines.push(`SEC_PG_PORT="${config.pgPort}"`);
      if (config.pgUser) lines.push(`SEC_PG_USER="${config.pgUser}"`);
      if (config.pgPassword) lines.push(`SEC_PG_PASSWORD="${config.pgPassword}"`);
      if (config.pgDatabase) lines.push(`SEC_PG_DATABASE="${config.pgDatabase}"`);
    }
  }

  return lines.join("\n") + "\n";
}

async function prompt(rl: ReturnType<typeof createInterface>, question: string): Promise<string> {
  return new Promise((resolve) => rl.question(question, resolve));
}

function addInitCommand(parent: Command): void {
  parent
    .command("init")
    .description("Interactive setup wizard for first-time configuration")
    .action(async () => {
      const globalOpts = parent.opts();
      if (globalOpts.json) {
        throw new Error("init command requires interactive mode (remove --json flag)");
      }

      await runCommand(async () => {
        const envPath = join(process.cwd(), ".env.local");
        if (existsSync(envPath)) {
          const existing = readFileSync(envPath, "utf-8");
          console.log(statusMessage("warn", ".env.local already exists:"));
          console.log(existing);
        }

        const rl = createInterface({ input: process.stdin, output: process.stdout });

        try {
          console.log("\nWelcome to @workglow/sec — SEC EDGAR data pipeline\n");

          // Step 1: DB type
          console.log("Step 1/4: Database type");
          const dbTypeInput = await prompt(rl, "  SQLite or PostgreSQL? [sqlite]: ");
          const dbType = dbTypeInput.toLowerCase() === "postgres" || dbTypeInput.toLowerCase() === "postgresql"
            ? "postgres" as const
            : "sqlite" as const;

          // Step 2: DB location
          console.log("\nStep 2/4: Database location");
          const dbFolder = (await prompt(rl, "  Database folder [~/.sec/data]: ")) || join(process.env.HOME ?? "~", ".sec", "data");
          const dbName = (await prompt(rl, "  Database name [edgar]: ")) || "edgar";

          // Step 3: Raw data folder
          console.log("\nStep 3/4: Raw data folder");
          const rawDataFolder = (await prompt(rl, "  Raw data folder [~/.sec/raw]: ")) || join(process.env.HOME ?? "~", ".sec", "raw");

          // Step 4: Postgres config (if needed)
          let pgUrl: string | undefined;
          let pgHost: string | undefined;
          let pgPort: string | undefined;
          let pgUser: string | undefined;
          let pgPassword: string | undefined;
          let pgDatabase: string | undefined;

          if (dbType === "postgres") {
            console.log("\nStep 3b: PostgreSQL connection");
            pgUrl = await prompt(rl, "  Connection string (or leave empty for individual params): ");
            if (!pgUrl) {
              pgHost = (await prompt(rl, "  Host [localhost]: ")) || "localhost";
              pgPort = (await prompt(rl, "  Port [5432]: ")) || "5432";
              pgUser = await prompt(rl, "  User: ");
              pgPassword = await prompt(rl, "  Password: ");
              pgDatabase = (await prompt(rl, "  Database [edgar]: ")) || "edgar";

          const config: InitConfig = {

          }
          if (!existsSync(rawDataFolder)) {
            mkdirSync(rawDataFolder, { recursive: true });
            console.log(statusMessage("success", `Created ${rawDataFolder}`));
          }

          writeFileSync(envPath, envContent);
          console.log(statusMessage("success", "Created .env.local"));

          // DB setup delegated to db setup command logic

// src/cli/queries/EntityQuery.ts
import { globalServiceRegistry } from "@workglow/util";
import { ENTITY_REPOSITORY_TOKEN } from "../../storage/entity/EntitySchema";

interface EntityQueryParams {
  readonly search?: string;
  readonly cik?: number;
  readonly sic?: number;
  readonly state?: string;
  readonly limit?: number;
  readonly offset?: number;
  readonly sort?: string;
}

interface QueryResult<T> {
  readonly rows: T[];
  readonly total: number;
}

async function queryEntities(params: EntityQueryParams): Promise<QueryResult<Record<string, unknown>>> {
  const repo = globalServiceRegistry.getService(ENTITY_REPOSITORY_TOKEN);
  // Get all entities — implementer should optimize with repo-specific query methods
  let all = await repo.getAll();

  if (params.cik !== undefined) {
    all = all.filter((e: Record<string, unknown>) => e.cik === params.cik);
  }
  if (params.sic !== undefined) {
    all = all.filter((e: Record<string, unknown>) => e.sic === params.sic);
  }
  if (params.state) {
    all = all.filter((e: Record<string, unknown>) => e.state_incorporation === params.state);
  }
  if (params.search) {
    const term = params.search.toLowerCase();
    all = all.filter((e: Record<string, unknown>) =>
      String(e.name ?? "").toLowerCase().includes(term)
    );
  }

  const total = all.length;
  const offset = params.offset ?? 0;
  const limit = params.limit ?? 25;
  const rows = all.slice(offset, offset + limit);

  return { rows, total };
}

```typescript
// src/cli/queries/FilingQuery.ts
import { globalServiceRegistry } from "@workglow/util";
import { FILING_REPOSITORY_TOKEN } from "../../storage/filing/FilingSchema";
import type { QueryResult } from "./EntityQuery";

interface FilingQueryParams {
  readonly search?: string;
  readonly cik?: number;
  readonly form?: string;
  readonly after?: string;
  readonly before?: string;
  readonly limit?: number;
  readonly offset?: number;
}

async function queryFilings(params: FilingQueryParams): Promise<QueryResult<Record<string, unknown>>> {
  const repo = globalServiceRegistry.getService(FILING_REPOSITORY_TOKEN);
  let all = await repo.getAll();

  if (params.cik !== undefined) {
    all = all.filter((f: Record<string, unknown>) => f.cik === params.cik);
  }
  if (params.form) {
    all = all.filter((f: Record<string, unknown>) => f.form === params.form);
  }
  if (params.after) {
    all = all.filter((f: Record<string, unknown>) => String(f.filing_date) >= params.after!);
  }
  if (params.before) {
    all = all.filter((f: Record<string, unknown>) => String(f.filing_date) <= params.before!);
  }
  if (params.search) {
    const term = params.search.toLowerCase();
    all = all.filter((f: Record<string, unknown>) =>
      String(f.primary_doc_description ?? "").toLowerCase().includes(term)
    );
  }

  const total = all.length;
  const offset = params.offset ?? 0;
  const limit = params.limit ?? 25;
  const rows = all.slice(offset, offset + limit);

  return { rows, total };
}

export { queryFilings, type FilingQueryParams };
```

import { globalServiceRegistry } from "@workglow/util";
import { ENTITY_REPOSITORY_TOKEN } from "../../storage/entity/EntitySchema";

  readonly filingCount: number;
  readonly factsCount: number;
  readonly processedSubmissions: number;
  readonly processedFacts: number;
  readonly processedFilings: number;

async function getDbStatus(): Promise<DbStatusResult> {
  const entityRepo = globalServiceRegistry.getService(ENTITY_REPOSITORY_TOKEN);
  const filingRepo = globalServiceRegistry.getService(FILING_REPOSITORY_TOKEN);
  const factsRepo = globalServiceRegistry.getService(COMPANY_FACTS_REPOSITORY_TOKEN);

  const [entities, filings, facts, procSubs, procFacts, procFilings] = await Promise.all([
    entityRepo.getAll(),
    filingRepo.getAll(),
    factsRepo.getAll(),
    procSubRepo.getAll(),
    procFactsRepo.getAll(),
    procFilingsRepo.getAll(),
  ]);

  return {
    entityCount: entities.length,
    filingCount: filings.length,
    factsCount: facts.length,
    processedSubmissions: procSubs.length,
    processedFacts: procFacts.length,
    processedFilings: procFilings.length,
  };
}
